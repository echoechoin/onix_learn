# 任务及上下文

## 任务

任务就是进程或者线程，协程，就是一个执行流；

- 程序入口地址
- 堆栈 - 内核栈
- 寄存器信息

## 手工创建一个任务

```c
#define PAGE_SZIE 0x1000

typedef struct task_t {
    u32 *stack; // 内核栈
} task_t;

typedef struct task_frame_t {
    u32 edi;
    u32 esi;
    u32 ebx;
    u32 ebp;
    u32 eip;
} task_frame_t;

task_t *task_a = 0x1000; // 将0x1000 作为存储task的上下文，size为PAGE_SIZE
task_t *task_b = 0x2000; // 将0x2000 作为存储task的上下文，size为PAGE_SIZE

// 栈指向内存页顶部
task_a->stack = task_a + PAGE_SIZE;

// 将 frame 压入栈中
task_frame_t *frame = task_a->stack;
task_a->stack -= sizeof(task_frame);

// 设置ip指针
frame->eip = thread_function;
```

## schedue 调度任务

1. 切换栈指针;
2. 修改eip;

实现一个`schedue()`函数切换任务

```c
// 获取当前任务
task_t *running_task()
{
    asm volatile(
        "movl %esp, %eax\n"
    // 因为task == 0x1000, 所以直接将当前的栈指针和0xfffff000
    // 存到eax里就变成了函数的返回值
        "andl $0xfffff000, %eax\n"
    );
}

void schedule()
{
    // 获取当前任务
    task_t *current = running_task();
    // 切换人物
    task_t *next = current == task_a ? task_b : task_a;
    task_switch(next);
}
```

### task_switch 的实现

切换上下文时需要切换的寄存器：

- ebx
- esi
- edi
- ebp
- esp

```s
global task_switch
task_switch:
    ; 1.1 保存栈帧到当前栈中
    push ebp
    mov ebp, esp

    ; 1.2 将 ebx esi edi 存放到当前栈中
    push ebx
    push esi
    push edi

    ; 1.3 获取当前task; 保存当前栈指针
    mov eax, esp
    and eax, 0xfffff000
    mov [eax], esp

    ; 2.1 获取 next task; 将next栈指针赋值给esp
    mov eax, [ebp + 8]; next
    mov esp, [eax]

    ; 2.2 pop next task 的 ebx esi edi ebp
    pop edi
    pop esi
    pop ebx
    pop ebp

    ; 2.3 恢复eip（因为此时栈中只剩下eip了）
    ret
```
